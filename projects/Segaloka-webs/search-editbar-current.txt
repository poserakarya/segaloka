"use client";

import {
  CalendarDays,
  MapPin,
  Search,
  UsersRound,
} from "lucide-react";

import {
  useMemo,
  useState,
} from "react";

import {
  useRouter,
} from "next/navigation";


type Props = {
  category: string;
  initialOrigin: string;
  initialDestination: string;
  initialTravelers: string;
  initialDate: string;
};


type Panel =
  | "origin"
  | "destination"
  | "traveler"
  | "date"
  | null;


const origins = [
  "Jakarta",
  "Makassar",
  "Surabaya",
  "Medan",
  "Bandung",
  "Yogyakarta",
  "Semarang",
  "Balikpapan",
];


const umrahTypes = [
  "Hemat",
  "Regular",
  "Plus",
  "First",
];


const categoryOptions: Record<string, string[]> = {
  Umrah: umrahTypes,

  Haji: [
    "Haji Reguler",
    "Haji Plus",
    "Haji Khusus",
  ],

  "Halal Tour": [
    "Jepang",
    "Korea Selatan",
    "Turkiye",
    "Dubai",
    "Eropa",
  ],

  "Tour Domestic": [
    "Bali",
    "Lombok",
    "Labuan Bajo",
    "Yogyakarta",
    "Raja Ampat",
  ],

  "Tour International": [
    "Jepang",
    "Korea Selatan",
    "Turkiye",
    "Dubai",
    "Singapura",
  ],

  SegaDeals: [
    "Umrah",
    "Halal Tour",
    "Tour Domestic",
    "Tour International",
  ],
};


function parseDate(
  value: string,
) {
  if (!value) {
    return "";
  }

  return value;
}


function formatDate(
  value: string,
) {
  if (!value) {
    return "Pilih tanggal";
  }

  const date =
    new Date(
      `${value}T00:00:00`,
    );

  if (
    Number.isNaN(
      date.getTime(),
    )
  ) {
    return value;
  }

  return new Intl.DateTimeFormat(
    "id-ID",
    {
      day: "numeric",
      month: "short",
      year: "numeric",
    },
  ).format(date);
}


export function SearchEditBar({
  category,
  initialOrigin,
  initialDestination,
  initialTravelers,
  initialDate,
}: Props) {

  const router =
    useRouter();


  const [panel, setPanel] =
    useState<Panel>(
      null,
    );


  const [origin, setOrigin] =
    useState(
      initialOrigin ||
      "Jakarta",
    );


  const initialUmrahType =
    umrahTypes.includes(
      initialDestination,
    )
      ? initialDestination
      : "Hemat";


  const [
    destination,
    setDestination,
  ] =
    useState(
      category === "Umrah"
        ? initialUmrahType
        : initialDestination,
    );


  const [
    travelers,
    setTravelers,
  ] =
    useState(
      Math.max(
        1,
        Number(
          initialTravelers ||
          2,
        ),
      ),
    );


  const [
    selectedDate,
    setSelectedDate,
  ] =
    useState(
      parseDate(
        initialDate,
      ),
    );


  const options =
    useMemo(
      () =>
        categoryOptions[
          category
        ] || [],
      [
        category,
      ],
    );


  const canSearch =
    origin.trim().length > 0 &&
    destination.trim().length > 0 &&
    travelers > 0 &&
    selectedDate.length > 0;


  function submitSearch() {

    if (!canSearch) {
      return;
    }

    const params =
      new URLSearchParams();

    params.set(
      "category",
      category,
    );

    params.set(
      "origin",
      origin,
    );

    params.set(
      "destination",
      destination,
    );

    params.set(
      "travelers",
      String(
        travelers,
      ),
    );

    params.set(
      "date",
      selectedDate,
    );

    setPanel(null);

    router.push(
      `/search?${params.toString()}`,
    );
  }


  return (
    <div className="search-edit-bar">


      <div className="search-edit-control">

        <button
          type="button"
          className="search-edit-field"
          onClick={() =>
            setPanel(
              panel === "origin"
                ? null
                : "origin",
            )
          }
        >
          <MapPin size={18} />

          <span>
            <small>Dari</small>

            <strong>
              {origin}
            </strong>
          </span>
        </button>


        {panel === "origin" && (

          <div className="search-edit-dropdown">

            <strong>
              Kota keberangkatan
            </strong>

            <div className="search-edit-option-list">

              {origins.map(
                (value) => (

                  <button
                    key={value}
                    type="button"
                    className={
                      value === origin
                        ? "active"
                        : ""
                    }
                    onClick={() => {
                      setOrigin(
                        value,
                      );

                      setPanel(
                        null,
                      );
                    }}
                  >
                    {value}
                  </button>

                ),
              )}

            </div>

          </div>

        )}

      </div>


      <div className="search-edit-control">

        <button
          type="button"
          className="search-edit-field"
          onClick={() =>
            setPanel(
              panel ===
                "destination"
                ? null
                : "destination",
            )
          }
        >

          <MapPin size={18} />

          <span>

            <small>
              {category === "Umrah"
                ? "Tujuan Umrah"
                : `Tujuan ${category}`}
            </small>

            <strong>
              {destination ||
                "Pilih tujuan"}
            </strong>

          </span>

        </button>


        {panel ===
          "destination" && (

          <div className="search-edit-dropdown search-edit-destination">

            <strong>
              {category === "Umrah"
                ? "Pilih Paket Umrah"
                : "Pilih Tujuan"}
            </strong>


            {category ===
            "Umrah" ? (

              <div className="search-umrah-options">

                {umrahTypes.map(
                  (value) => (

                    <button
                      key={value}
                      type="button"
                      className={
                        value ===
                        destination
                          ? "active"
                          : ""
                      }
                      onClick={() => {
                        setDestination(
                          value,
                        );

                        setPanel(
                          null,
                        );
                      }}
                    >
                      {value}
                    </button>

                  ),
                )}

              </div>

            ) : (

              <div className="search-edit-option-list">

                {options.map(
                  (value) => (

                    <button
                      key={value}
                      type="button"
                      className={
                        value ===
                        destination
                          ? "active"
                          : ""
                      }
                      onClick={() => {
                        setDestination(
                          value,
                        );

                        setPanel(
                          null,
                        );
                      }}
                    >
                      {value}
                    </button>

                  ),
                )}

              </div>

            )}

          </div>

        )}

      </div>


      <div className="search-edit-control">

        <button
          type="button"
          className="search-edit-field"
          onClick={() =>
            setPanel(
              panel === "date"
                ? null
                : "date",
            )
          }
        >

          <CalendarDays
            size={18}
          />

          <span>

            <small>
              Tanggal
            </small>

            <strong>
              {formatDate(
                selectedDate,
              )}
            </strong>

          </span>

        </button>


        {panel === "date" && (

          <div className="search-edit-dropdown search-edit-date">

            <strong>
              Pilih tanggal
            </strong>

            <input
              type="date"
              value={
                selectedDate
              }
              min={
                new Date()
                  .toISOString()
                  .slice(
                    0,
                    10,
                  )
              }
              onChange={(event) => {
                setSelectedDate(
                  event.target.value,
                );

                if (
                  event.target.value
                ) {
                  setPanel(
                    null,
                  );
                }
              }}
            />

          </div>

        )}

      </div>


      <div className="search-edit-control">

        <button
          type="button"
          className="search-edit-field"
          onClick={() =>
            setPanel(
              panel ===
                "traveler"
                ? null
                : "traveler",
            )
          }
        >

          <UsersRound
            size={18}
          />

          <span>

            <small>
              Traveler
            </small>

            <strong>
              {travelers} orang
            </strong>

          </span>

        </button>


        {panel ===
          "traveler" && (

          <div className="search-edit-dropdown search-edit-traveler">

            <div>

              <strong>
                Traveler
              </strong>

              <small>
                Jumlah peserta
              </small>

            </div>


            <div className="search-edit-counter">

              <button
                type="button"
                disabled={
                  travelers <= 1
                }
                onClick={() =>
                  setTravelers(
                    (value) =>
                      Math.max(
                        1,
                        value - 1,
                      ),
                  )
                }
              >
                −
              </button>


              <strong>
                {travelers}
              </strong>


              <button
                type="button"
                onClick={() =>
                  setTravelers(
                    (value) =>
                      value + 1,
                  )
                }
              >
                +
              </button>

            </div>

          </div>

        )}

      </div>


      <button
        type="button"
        className="search-edit-submit"
        disabled={!canSearch}
        onClick={
          submitSearch
        }
      >

        <Search size={18} />

        Cari

      </button>

    </div>
  );
}
