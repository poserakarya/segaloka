function getMonthNames(locale: string) {
  const formatter = new Intl.DateTimeFormat(locale, {
    month: "long",
  });

  return Array.from(
    { length: 12 },
    (_, month) =>
      formatter.format(
        new Date(2026, month, 1),
      ),
  );
}


function getDayNames(locale: string) {
  const formatter = new Intl.DateTimeFormat(locale, {
    weekday: "short",
  });

  const sunday = new Date(2026, 0, 4);

  return Array.from(
    { length: 7 },
    (_, index) => {
      const date = new Date(sunday);

      date.setDate(
        sunday.getDate() + index,
      );

      return formatter.format(date);
    },
  );
}

function formatDate(
  date: Date | null,
  locale: string,
  emptyLabel: string,
) {
  if (!date) {
    return emptyLabel;
  }

  return new Intl.DateTimeFormat(locale, {
    day: "numeric",
    month: "short",
    year: "numeric",
  }).format(date);
}


function getCalendarDays(year: number, month: number) {
  const firstDay = new Date(year, month, 1).getDay();
  const numberOfDays = new Date(
    year,
    month + 1,
    0,
  ).getDate();

  const result: Array<number | null> = [];

  for (let index = 0; index < firstDay; index += 1) {
    result.push(null);
  }

  for (let day = 1; day <= numberOfDays; day += 1) {
    result.push(day);
  }

  while (result.length % 7 !== 0) {
    result.push(null);
  }

  return result;
}


function CalendarMonth({
  year,
  month,
  selectedDate,
  onSelect,
  locale,
}: {
  year: number;
  month: number;
  selectedDate: Date | null;
  onSelect: (date: Date) => void;
  locale: string;
}) {
  const days = getCalendarDays(year, month);
  const monthNames = getMonthNames(locale);
  const dayNames = getDayNames(locale);

  const today = new Date();
  today.setHours(0, 0, 0, 0);

  return (
    <div className="hero-calendar-month">
      <h4>
        {monthNames[month]} {year}
      </h4>

      <div className="hero-calendar-weekdays">
        {dayNames.map((name, index) => (
          <span
            key={name}
            className={index === 0 ? "sunday" : ""}
          >
            {name}
          </span>
        ))}
      </div>

      <div className="hero-calendar-days">
        {days.map((day, index) => {
          if (!day) {
            return (
              <span
                key={`empty-${index}`}
                className="empty"
              />
            );
          }

          const candidate = new Date(
            year,
            month,
            day,
          );

          candidate.setHours(0, 0, 0, 0);

          const disabled = candidate < today;

          const selected =
            selectedDate &&
            candidate.getFullYear() ===
              selectedDate.getFullYear() &&
            candidate.getMonth() ===
              selectedDate.getMonth() &&
            candidate.getDate() ===
              selectedDate.getDate();

          return (
            <button
              key={day}
              type="button"
              disabled={disabled}
              className={selected ? "selected" : ""}
              onClick={() => {
                if (!disabled) {
                  onSelect(candidate);
                }
              }}
            >
              {day}
            </button>
          );
        })}
      </div>
    </div>
  );
}

    }

    params.set(

        ? segaOriginCity
        : origin,
    );

    if (destination) {

        "year",
        String(hajiYear),
      );
    }

      params.set(
        "budget",
        budget,
      );

function pad(value: number) {
  return String(value).padStart(2, "0");
}

function toDateValue(
  year: number,
  month: number,
  day: number,
) {
  return `${year}-${pad(month + 1)}-${pad(day)}`;
}

function parseDateValue(value: string) {
  if (!value) {
    return null;
  }

  const parts = value
    .split("-")
    .map(Number);

  if (parts.length !== 3) {
    return null;
  }

  const [year, month, day] = parts;

  if (
    !year ||
    !month ||
    !day
  ) {
    return null;
  }

  return new Date(
    year,
    month - 1,
    day,
  );
}

function sameDay(
  first: Date,
  second: Date,
) {
  return (
    first.getFullYear() === second.getFullYear() &&
    first.getMonth() === second.getMonth() &&
    first.getDate() === second.getDate()
  );
}

function buildMonthDays(
  year: number,
  month: number,
) {
  const firstDay =
    new Date(
      year,
      month,
      1,
    ).getDay();

  const totalDays =
    new Date(
      year,
      month + 1,
      0,
    ).getDate();

  const result: Array<number | null> = [];

  for (
    let index = 0;
    index < firstDay;
    index += 1
  ) {
    result.push(null);
  }

  for (
    let day = 1;
    day <= totalDays;
    day += 1
  ) {
    result.push(day);
  }

  while (
    result.length % 7 !== 0
  ) {
    result.push(null);
  }

  return result;
}

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from "react";

import {
  type CurrencyCode,
  type LanguageCode,
} from "@/data/preferences";

type MarketplacePreferencesContextValue = {
  language: LanguageCode;
  currency: CurrencyCode;
  setLanguage: (language: LanguageCode) => void;
  setCurrency: (currency: CurrencyCode) => void;
  locale: string;
  direction: "ltr" | "rtl";
  formatCurrency: (amountInIdr: number) => string;
  convertFromIdr: (amountInIdr: number) => number;
};

const STORAGE_KEY = "segaloka-marketplace-preferences";

const languageLocale: Record<LanguageCode, string> = {
  ID: "id-ID",
  EN: "en-US",
  AR: "ar-SA",
};

/*
 * UI prototype rates.
 *
 * Semua nominal produk saat ini tetap memakai IDR sebagai base amount.
 * Rate ini hanya untuk DISPLAY marketplace.
 *
 * Saat API/backend kurs diaktifkan nanti, ganti sumber rate ini dengan
 * rate service dan simpan rate snapshot pada transaksi.
 */
const displayRates: Record<CurrencyCode, number> = {
  IDR: 1,
  USD: 1 / 16500,
  SAR: 1 / 4400,
};

const MarketplacePreferencesContext =
  createContext<MarketplacePreferencesContextValue | null>(null);

function isLanguageCode(value: unknown): value is LanguageCode {
  return value === "ID" || value === "EN" || value === "AR";
}

function isCurrencyCode(value: unknown): value is CurrencyCode {
  return value === "IDR" || value === "USD" || value === "SAR";
}

export function MarketplacePreferencesProvider({
  children,
}: {
  children: ReactNode;
}) {
  const [language, setLanguageState] =
    useState<LanguageCode>("ID");

  const [currency, setCurrencyState] =
    useState<CurrencyCode>("IDR");

  const locale = languageLocale[language];
  const direction: "ltr" | "rtl" =
    language === "AR" ? "rtl" : "ltr";

  /*
   * Preference non-sensitif boleh dipersist.
   * Ini TIDAK digunakan untuk traveler/passport/payment data.
   */
  useEffect(() => {
    try {
      const stored = window.localStorage.getItem(STORAGE_KEY);

      if (!stored) {
        return;
      }

      const parsed = JSON.parse(stored) as {
        language?: unknown;
        currency?: unknown;
      };

      if (isLanguageCode(parsed.language)) {
        setLanguageState(parsed.language);
      }

      if (isCurrencyCode(parsed.currency)) {
        setCurrencyState(parsed.currency);
      }
    } catch {
      // Invalid preference storage should never break marketplace.
    }
  }, []);

  useEffect(() => {
    document.documentElement.lang =
      language === "ID"
        ? "id"
        : language === "EN"
          ? "en"
          : "ar";

    document.documentElement.dir = direction;
  }, [language, direction]);

  const persist = useCallback(
    (
      nextLanguage: LanguageCode,
      nextCurrency: CurrencyCode,
    ) => {
      try {
        window.localStorage.setItem(
          STORAGE_KEY,
          JSON.stringify({
            language: nextLanguage,
            currency: nextCurrency,
          }),
        );
      } catch {
        // Preference persistence is optional.
      }
    },
    [],
  );

  const setLanguage = useCallback(
    (nextLanguage: LanguageCode) => {
      setLanguageState(nextLanguage);
      persist(nextLanguage, currency);
    },
    [currency, persist],
  );

  const setCurrency = useCallback(
    (nextCurrency: CurrencyCode) => {
      setCurrencyState(nextCurrency);
      persist(language, nextCurrency);
    },
    [language, persist],
  );

  const convertFromIdr = useCallback(
    (amountInIdr: number) => {
      return amountInIdr * displayRates[currency];
    },
    [currency],
  );

  const formatCurrency = useCallback(
    (amountInIdr: number) => {
      const converted =
        amountInIdr * displayRates[currency];

      return new Intl.NumberFormat(locale, {
        style: "currency",
        currency,
        maximumFractionDigits:
          currency === "IDR" ? 0 : 2,
        minimumFractionDigits: 0,
      }).format(converted);
    },
    [currency, locale],
  );

  const value = useMemo(
    () => ({
      language,
      currency,
      setLanguage,
      setCurrency,
      locale,
      direction,
      formatCurrency,
      convertFromIdr,
    }),
    [
      language,
      currency,
      setLanguage,
      setCurrency,
      locale,
      direction,
      formatCurrency,
      convertFromIdr,
    ],
  );

  return (
    <MarketplacePreferencesContext.Provider value={value}>
      {children}
    </MarketplacePreferencesContext.Provider>
  );
}

export function useMarketplacePreferences() {
  const context = useContext(
    MarketplacePreferencesContext,
  );

  if (!context) {
    throw new Error(
      "useMarketplacePreferences harus digunakan di dalam MarketplacePreferencesProvider.",
    );
  }

  return context;
}
