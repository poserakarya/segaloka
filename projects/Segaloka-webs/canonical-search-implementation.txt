function getMonthNames(locale: string) {
  const formatter = new Intl.DateTimeFormat(locale, {
    month: "long",
  });

  return Array.from(
    { length: 12 },
    (_, month) =>
      formatter.format(
        new Date(2026, month, 1),
      ),
  );
}


function getDayNames(locale: string) {
  const formatter = new Intl.DateTimeFormat(locale, {
    weekday: "short",
  });

  const sunday = new Date(2026, 0, 4);

  return Array.from(
    { length: 7 },
    (_, index) => {
      const date = new Date(sunday);

      date.setDate(
        sunday.getDate() + index,
      );

      return formatter.format(date);
    },
  );
}

function formatDate(
  date: Date | null,
  locale: string,
  emptyLabel: string,
) {
  if (!date) {
    return emptyLabel;
  }

  return new Intl.DateTimeFormat(locale, {
    day: "numeric",
    month: "short",
    year: "numeric",
  }).format(date);
}


function getCalendarDays(year: number, month: number) {
  const firstDay = new Date(year, month, 1).getDay();
  const numberOfDays = new Date(
    year,
    month + 1,
    0,
  ).getDate();

  const result: Array<number | null> = [];

  for (let index = 0; index < firstDay; index += 1) {
    result.push(null);
  }

  for (let day = 1; day <= numberOfDays; day += 1) {
    result.push(day);
  }

  while (result.length % 7 !== 0) {
    result.push(null);
  }

  return result;
}


function CalendarMonth({
  year,
  month,
  selectedDate,
  onSelect,
  locale,
}: {
  year: number;
  month: number;
  selectedDate: Date | null;
  onSelect: (date: Date) => void;
  locale: string;
}) {
  const days = getCalendarDays(year, month);
  const monthNames = getMonthNames(locale);
  const dayNames = getDayNames(locale);

  const today = new Date();
  today.setHours(0, 0, 0, 0);

  return (
    <div className="hero-calendar-month">
      <h4>
        {monthNames[month]} {year}
      </h4>

      <div className="hero-calendar-weekdays">
        {dayNames.map((name, index) => (
          <span
            key={name}
            className={index === 0 ? "sunday" : ""}
          >
            {name}
          </span>
        ))}
      </div>

      <div className="hero-calendar-days">
        {days.map((day, index) => {
          if (!day) {
            return (
              <span
                key={`empty-${index}`}
                className="empty"
              />
            );
          }

          const candidate = new Date(
            year,
            month,
            day,
          );

          candidate.setHours(0, 0, 0, 0);

          const disabled = candidate < today;

          const selected =
            selectedDate &&
            candidate.getFullYear() ===
              selectedDate.getFullYear() &&
            candidate.getMonth() ===
              selectedDate.getMonth() &&
            candidate.getDate() ===
              selectedDate.getDate();

          return (
            <button
              key={day}
              type="button"
              disabled={disabled}
              className={selected ? "selected" : ""}
              onClick={() => {
                if (!disabled) {
                  onSelect(candidate);
                }
              }}
            >
              {day}
            </button>
          );
        })}
      </div>
    </div>
  );
}

    }

    params.set(

        ? segaOriginCity
        : origin,
    );

    if (destination) {

        "year",
        String(hajiYear),
      );
    }

      params.set(
        "budget",
        budget,
      );

function pad(value: number) {
  return String(value).padStart(2, "0");
}

function toDateValue(
  year: number,
  month: number,
  day: number,
) {
  return `${year}-${pad(month + 1)}-${pad(day)}`;
}

function parseDateValue(value: string) {
  if (!value) {
    return null;
  }

  const parts = value
    .split("-")
    .map(Number);

  if (parts.length !== 3) {
    return null;
  }

  const [year, month, day] = parts;

  if (
    !year ||
    !month ||
    !day
  ) {
    return null;
  }

  return new Date(
    year,
    month - 1,
    day,
  );
}

function sameDay(
  first: Date,
  second: Date,
) {
  return (
    first.getFullYear() === second.getFullYear() &&
    first.getMonth() === second.getMonth() &&
    first.getDate() === second.getDate()
  );
}

function buildMonthDays(
  year: number,
  month: number,
) {
  const firstDay =
    new Date(
      year,
      month,
      1,
    ).getDay();

  const totalDays =
    new Date(
      year,
      month + 1,
      0,
    ).getDate();

  const result: Array<number | null> = [];

  for (
    let index = 0;
    index < firstDay;
    index += 1
  ) {
    result.push(null);
  }

  for (
    let day = 1;
    day <= totalDays;
    day += 1
  ) {
    result.push(day);
  }

  while (
    result.length % 7 !== 0
  ) {
    result.push(null);
  }

  return result;
}

                  value={currentYear}
                  onChange={(event) =>
                    setCalendarMonth(
                      new Date(
                        Number(
                          event.target.value,
                        ),
                        currentMonth,
                        1,
                      ),
                    )
                  }
                >
                  {Array.from(
                    {
                      length: 4,
                    },
                    (_, index) =>
                      today.getFullYear() +
                      index,
                  ).map(
                    (year) => (
                      <option
                        key={year}
                        value={year}
                      >
                        {year}
                      </option>
                    ),
                  )}
                </select>
              </div>

              <button
                type="button"
                aria-label="Next month"
                onClick={() =>
                  setCalendarMonth(
                    new Date(
                      currentYear,
                      currentMonth + 1,
                      1,
                    ),
                  )
                }
              >
                <ChevronRight size={18} />
              </button>
            </div>

            <div className="search-unified-calendar-grid">
              {renderMonth(
                currentYear,
                currentMonth,
                monthOneDays,
              )}

              {renderMonth(
                nextMonth.getFullYear(),
                nextMonth.getMonth(),
                monthTwoDays,
              )}
            </div>
          </div>
        )}
      </div>

      <div className="search-edit-control">
        <button
          type="button"
          className="search-edit-field"
          aria-expanded={
            panel === "traveler"
          }
          onClick={() =>
            togglePanel(
              "traveler",
            )
          }
        >
          <UsersRound size={18} />

          <span>
            <small>
              {t.traveler}
            </small>

            <strong>
              {travelerTotal}{" "}
              {t.person}
            </strong>
          </span>
        </button>

        {panel ===
          "traveler" && (
          <div className="search-edit-dropdown search-unified-dropdown search-unified-traveler">
            <div className="search-unified-dropdown-head">
              <div>
                <strong>
                  {t.travelerTitle}
                </strong>

                <small>
                  {
                    t.travelerDescription
                  }
                </small>
              </div>

              <button
                type="button"
                className="search-unified-close"
                aria-label="Close"
                onClick={() =>
                  setPanel(null)
                }
              >
                <X size={18} />
              </button>
            </div>

            {(
              [
                [
                  "adult",
                  t.adult,
                  t.adultDescription,
                ],
                [
                  "child",
                  t.child,
                  t.childDescription,
                ],
                [
                  "infant",
                  t.infant,
                  t.infantDescription,
                ],
              ] as const
            ).map(
              ([
                key,
                title,
                description,
              ]) => (
                <div
                  className="search-unified-traveler-row"
                  key={key}
                >
                  <div>
                    <strong>
                      {title}
                    </strong>

                    <small>
                      {description}
                    </small>
                  </div>

                  <div className="search-unified-counter">
                    <button
                      type="button"
                      disabled={
                        travelers[key] <=
                          0 ||
                        travelerTotal <= 1
                      }
                      onClick={() =>
                        updateTraveler(
                          key,
                          -1,
                        )
                      }
                    >
                      <Minus size={16} />
                    </button>

                    <strong>
                      {travelers[key]}
                    </strong>

                    <button
                      type="button"
                      onClick={() =>
                        updateTraveler(
                          key,
                          1,
                        )
                      }
                    >
                      <Plus size={16} />
                    </button>
                  </div>
                </div>
              ),
            )}

            <button
              type="button"
              className="search-unified-done"
              onClick={() =>
                setPanel(null)
              }
            >
              {t.done}
            </button>
          </div>
        )}
      </div>

      <button
        type="button"
        className="search-edit-submit"
        disabled={!canSearch}
        onClick={submitSearch}
      >
        <Search size={18} />
        {t.search}
      </button>
    </div>
  );
}
